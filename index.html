import torch
import torch.nn as nn

class SimpleCNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.conv1 = nn.Conv2d(3, 64, 3)
        self.fc = nn.Linear(64 * 16 * 16, 10)
    
    def forward(self, x):
        x = torch.relu(self.conv1(x))
        x = x.view(-1, 64 * 16 * 16)
        return self.fc(x)

# 训练模型
model = SimpleCNN()
optimizer = torch.optim.Adam(model.parameters())
